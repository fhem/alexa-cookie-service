# Beispiel: externen Refresh triggern und den Cookie direkt im Exportverzeichnis speichern
# Anpassung der URLs/Pfade an deine FHEM-Umgebung ist nötig.

defmod AlexaCookieRefresh at +*06:00:00 {
  my $name = 'EchoWohnzimmer';;
  my $hash = $defs{$name};;
  my $save_name = $hash->{NR} . 'result.json';;
  setKeyValue('alexa_cookie_service_export_name', $save_name);;
  my $token = 'change-me';;
  my $refresh_raw = qx(curl -sS -X POST -H "x-auth-token: $token" -w "\\n%{http_code}" "http://127.0.0.1:58080/api/cookie/refresh?save=$save_name" 2>&1);;
  return 'refresh failed: empty response' if !defined($refresh_raw) || $refresh_raw eq '';;
  $refresh_raw =~ s/\r//g;;
  my ($refresh_body, $refresh_code) = $refresh_raw =~ /\A(.*)\n(\d{3})\z/s;;
  return "refresh failed: could not parse HTTP status from response: $refresh_raw"
    if !defined($refresh_code);;
  return "refresh failed: http=$refresh_code body=$refresh_body"
    if $refresh_code ne '200';;
  return 'refresh failed: empty body' if $refresh_body eq '';;
  echodevice_NPMWaitForCookie($hash);;
  Log 1, "AlexaCookieRefresh executed";;
}
